# silentcipher codec-robust fine-tune — Python dependencies
# Pinned to the versions Sony's silentcipher training code was
# validated against + the codec-augmentation stack (Encodec Option A
# primary; librosa/soundfile fallback for Option B).
#
# Install with:
#     python -m venv .venv
#     source .venv/bin/activate  # or .venv\Scripts\activate on Windows
#     pip install -r requirements.txt
#
# Not committed to the workspace Cargo.lock — this venv lives inside
# tools/train-silentcipher-codec-robust/ only and does not affect the
# release matrix or CI.

# under-utilises 5090's tensor cores.

# reference + encoder/decoder module definitions we fine-tune against.

# Hugging Face hub — Sony's checkpoint is hosted here.
huggingface_hub>=0.24.0

# Audio processing.
librosa>=0.10.2

numpy>=1.26.0

# Encodec (Meta FAIR, MIT) — Option A augmentation path.
encodec>=0.1.1

# Config file parsing (night1.yaml / night2.yaml).
pyyaml>=6.0.1

# Progress + logging.
tqdm>=4.66.0

# Optional: tensorboard for the between-night validation dashboard.

tensorboard>=2.16.0

# psutil — cross-platform process-priority downgrade + system inspection.
